\section{Method Excerpt}
Let $\mathbf{U}\in\mathbb{R}^{D\times2}$ have orthonormal columns.
For each historical memory, retain
$\mathbf{z}_i=\mathbf{U}^{\top}\mathbf{v}_i$.
For cluster $\mathcal{C}_k$, retain its visual sum $\mathbf{S}_k$,
anchor $\mathbf{a}_k=\mathbf{S}_k/|\mathcal{C}_k|$, and the derived
residual
\begin{equation}
    \mathbf{R}_k=\mathbf{S}_k-
    \mathbf{U}\sum_{i\in\mathcal{C}_k}\mathbf{z}_i.
\end{equation}
The reconstruction for partial member reassignment is
\begin{equation}
\label{eq:vamo-leaf-reconstruction}
    \hat{\mathbf{v}}_i
    =\mathbf{U}\mathbf{z}_i+
     \frac{\mathbf{R}_k}{|\mathcal{C}_k|}.
\end{equation}

For updated clusters $\mathcal{C}'_k$ and newly arrived indices
$\mathcal{N}$, the unified update is
\begin{equation}
\label{eq:vamo-unified-update}
\begin{aligned}
\mathbf{S}'_k
=\sum_j\Bigg[
&\mathbf{U}
 \sum_{i\in\mathcal{C}_j\cap\mathcal{C}'_k}\mathbf{z}_i\\
&+\frac{|\mathcal{C}_j\cap\mathcal{C}'_k|}
 {|\mathcal{C}_j|}\mathbf{R}_j
\Bigg]
+\sum_{i\in\mathcal{C}'_k\cap\mathcal{N}}\mathbf{v}_i.
\end{aligned}
\end{equation}
